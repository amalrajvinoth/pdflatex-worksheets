\newcommand{\genAdd}[1][]{%
  \pgfkeys{/prob, tens min=1, tens max=4, ones min=0, ones max=9, tens2 min=1, tens2 max=4, ones2 min=0, ones2 max=9, carry=0}%
  \ifx#1\@empty\else\edef\@tempa{#1}\pgfkeys{/prob, \@tempa}\fi
  \pgfkeysgetvalue{/prob/carry}{\carryval}%
  \pgfkeysgetvalue{/prob/tens min}{\tmin}\pgfkeysgetvalue{/prob/tens max}{\tmax}%
  \pgfkeysgetvalue{/prob/ones min}{\omin}\pgfkeysgetvalue{/prob/ones max}{\omax}%
  \pgfmathrandominteger{\ta}{\tmin}{\tmax}%
  \pgfmathrandominteger{\oa}{\omin}{\omax}%
  \ifnum\carryval=1 %
    \pgfmathtruncatemacro{\maxtb}{8-\ta}%
    \ifnum\maxtb<1\def\maxtb{1}\fi%
    \pgfmathrandominteger{\tb}{1}{\maxtb}%
    \pgfmathtruncatemacro{\minob}{10-\oa}%
    \ifnum\minob<\omin\def\minob{\omin}\fi
    \pgfmathtruncatemacro{\maxob}{9}%
    \ifnum\omax<\maxob\def\maxob{\omax}\fi
    \ifnum\minob>\maxob\def\minob{\maxob}\fi
    \pgfmathrandominteger{\ob}{\minob}{\maxob}%
  \else%
    \pgfmathtruncatemacro{\maxtb}{9-\ta}%
    \ifnum\maxtb<1\def\maxtb{1}\fi%
    \pgfmathrandominteger{\tb}{1}{\maxtb}%
    \pgfmathtruncatemacro{\maxob}{9-\oa}%
    \pgfmathrandominteger{\ob}{0}{\maxob}%
  \fi%
  \pgfmathtruncatemacro{\resulta}{\ta*10+\oa}%
  \pgfmathtruncatemacro{\resultb}{\tb*10+\ob}%
  \global\edef\problemnuma{\resulta}%
  \global\edef\problemnumb{\resultb}%
}

\newcommand{\genSub}[1][]{%
  \pgfkeys{/prob, tens min=2, tens max=9, ones min=0, ones max=9, borrow=0}%
  \ifx#1\@empty\else\edef\@tempa{#1}\pgfkeys{/prob, \@tempa}\fi
  \pgfkeysgetvalue{/prob/borrow}{\borrowval}%
  \pgfkeysgetvalue{/prob/tens min}{\tmin}\pgfkeysgetvalue{/prob/tens max}{\tmax}%
  \pgfkeysgetvalue{/prob/ones min}{\omin}\pgfkeysgetvalue{/prob/ones max}{\omax}%
  \pgfmathrandominteger{\ta}{\tmin}{\tmax}%
  \pgfmathrandominteger{\oa}{\omin}{\omax}%
  \ifnum\borrowval=1 %
    \pgfmathtruncatemacro{\maxtb}{\ta-2}%
    \ifnum\maxtb<1\def\maxtb{1}\fi%
    \pgfmathrandominteger{\tb}{1}{\maxtb}%
    \pgfmathtruncatemacro{\minob}{\oa+1}%
    \ifnum\minob>9\def\minob{9}\fi%
    \pgfmathrandominteger{\ob}{\minob}{9}%
  \else%
    \pgfmathtruncatemacro{\maxtb}{\ta-1}%
    \ifnum\maxtb<1\def\maxtb{1}\fi%
    \pgfmathrandominteger{\tb}{1}{\maxtb}%
    \ifnum\oa<1%
      \def\ob{0}%
    \else%
      \pgfmathrandominteger{\ob}{0}{\oa}%
    \fi%
  \fi%
  \pgfmathtruncatemacro{\resulta}{\ta*10+\oa}%
  \pgfmathtruncatemacro{\resultb}{\tb*10+\ob}%
  \global\edef\problemnuma{\resulta}%
  \global\edef\problemnumb{\resultb}%
}

\newcommand{\problem}[3]{%
  \noindent
  \begin{minipage}[t]{1in}%
    \centering
    \begin{tabular}[t]{r}%
      {\large #1}\\[\digitspace]%
      {\large #2~#3}\\[\linespace]%
      \hline
    \end{tabular}%
  \end{minipage}%
  \par\vspace{\problemspace}%
}

\newcommand{\addP}[2]{\problem{#1}{+}{#2}}
\newcommand{\subP}[2]{\problem{#1}{-}{#2}}

\newcommand{\genProblems}[2]{\foreach \i in {1,...,#2}{#1}}

\newcommand{\MathSheetMixed}[4]{%
  \fancyhead[C]{\large\textbf{Kids Math - Under 100 (#4)}}%
  \begin{multicols}{4}
    \genProblems{#1\addP{\problemnuma}{\problemnumb}}{9}
    \columnbreak
    \genProblems{#1\addP{\problemnuma}{\problemnumb}}{9}
    \columnbreak
    \genProblems{#2\subP{\problemnuma}{\problemnumb}}{9}
    \columnbreak
    \genProblems{#2\subP{\problemnuma}{\problemnumb}}{9}
  \end{multicols}
  \foreach \p in {1,...,#3}{\newpage
    \begin{multicols}{4}
      \genProblems{#1\addP{\problemnuma}{\problemnumb}}{9}
      \columnbreak
      \genProblems{#1\addP{\problemnuma}{\problemnumb}}{9}
      \columnbreak
      \genProblems{#2\subP{\problemnuma}{\problemnumb}}{9}
      \columnbreak
      \genProblems{#2\subP{\problemnuma}{\problemnumb}}{9}
    \end{multicols}
  }
  \newpage
}

\newcommand{\MakeMathSheet}[5]{%
  \global\edef\@MUAoptsA{#2}%
  \global\edef\@MUAoptsS{#3}%
  \ifnum\pdfstrcmp{#1}{add}=0
    \MathSheetAdd{\expandafter\genAdd\expandafter[\@MUAoptsA]}{#4}{#5}%
  \else\ifnum\pdfstrcmp{#1}{sub}=0
    \MathSheetSub{\expandafter\genSub\expandafter[\@MUAoptsS]}{#4}{#5}%
  \else
    \MathSheetMixed{\expandafter\genAdd\expandafter[\@MUAoptsA]}{\expandafter\genSub\expandafter[\@MUAoptsS]}{#4}{#5}%
  \fi\fi
}
